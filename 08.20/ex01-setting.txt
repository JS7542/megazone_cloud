# ################################################################################
# EKS 개발 환경 구성(로컬 환경)
# 운영체제: Windows 11 WSL Ubuntu-24.04
# ################################################################################

# ################################################################################
# awscli / kubectl / eksctl 설치
# ================================================================================
# 1. 패키지 업데이트 및 압축 해제 도구 설치
sudo apt update && sudo apt install -y unzip curl

# 2. awscli 설치 파일 다운로드 및 실행 AWS 로그인 환경 구성
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install
aws --version
rm -rf aws awscliv2.zip
aws configure

# 3. kubectl 최신 안정화 버전 다운로드 및 실행 권한 부여와 경로 이동
# 최신버전: curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
curl -LO "https://dl.k8s.io/release/v1.34.9/bin/linux/amd64/kubectl"
chmod +x kubectl
sudo mv kubectl /usr/local/bin/
kubectl version --client    # --client: 현재 로컬의 버전만 확인, 없을 경우 

# 4. eksctl 최신 버전 다운로드 및 압축 해제 및 실행 파일 이동
curl --silent --location "https://github.com/weaveworks/eksctl/releases/latest/download/eksctl_$(uname -s)_amd64.tar.gz" | tar xz -C /tmp
sudo mv /tmp/eksctl /usr/local/bin
eksctl version

# CLI 명령어 자동 완성 기능 추가
source <(kubectl completion bash)
echo "source <(kubectl completion bash)" >> ~/.bashrc
