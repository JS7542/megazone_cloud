이렇게 작성하면 어떤 책을 얼마에 팔았는지 알수가 없기에, 참조키를 이용한다.이러한 참조키는 이전에 만든 tbookinfo 에서 참조한다.




CREATE TABLE IF NOT EXISTS tsellinfo (
	idx			INT			AUTO_INCREMENT,
	book		INT 		NOT NULL 		COMMENT 'tbookinfo 에서 맺어진 참조 키 값',
    price		INT			NOT NULL		COMMENT '도서 판매가',
    salenum		INT			NOT NULL		COMMENT '도서 판매 수량',
    seldate		date		NOT NULL		COMMENT '도서 판매 일자',
    PRIMARY KEY(idx),
    CONSTRAINT fk_tbookinfo_idx FOREIGN KEY(book) REFERENCES tbookinfo (idx)
)COMMENT '도서 판매 정보';

DB_백업 쉘 스크립트
==================================================================================
#!/bin/bash

#날짜 포멧
DATE=$(date +%Y%m%d_%H%M)

#백업 파일명
BACK_FILE="mysql-backup-${DATE}.sql.gz"

#백업 진행및 용령절약을 위한 압축 진행
mysqldump -u std20 -pmelt7542 std20db | gzip > "/home/ubuntu/backup/${BACK_FILE}"

#s3 업로드
aws s3 cp ./backup/${BACK_FILE} s3://std20-mysql-backup-bucket/mysql/

#용량 최적화를 위해 업로드 후 원본 압축파일 삭제
rm -f ./backup/${BACK_FILE}

지금은 실습환경이기에, 경로를 모두 지정하였지만, 실제로 할때에는 가변적인 부분은 변수를 활용하여, 실행할때마다 입력하거나,
실행전 변수만 수정하게끔 하여야 한다.

수정후 코드
=======================================================================================
#!/bin/bash

#DB 및 환경 변수
#DB_USER=$1
#DB_PASS=$2									
#DB_NAME=$3
DB_USER="std20"
DB_PASS="melt7542"									# 실제 DB 비밀번호 입력
DB_NAME="std20db"									# 백업할 데이터베이스명 입력
BACKUP_DIR="backup"
BUCKET_NAME="std20-mysql-backup-bucket"				# 백업할 버킷 이름
BUCKET_DIR="mysql"									# 버킷 내부의 저장 디렉토리 이름

DATE=$(date +%Y%m%d_%H%M)							#날짜 포멧

BACK_FILE="${DB_NAME}-backup-${DATE}.sql.gz"		#백업 파일명

													#백업 진행및 용령절약을 위한 압축 진행
mysqldump -u ${DB_USER} -p${DB_PASS} ${DB_NAME} | gzip > "/home/ubuntu/${BACKUP_DIR}/${BACK_FILE}"

													#s3 업로드
aws s3 cp ./${BACKUP_DIR}/${BACK_FILE} s3://${BUCKET_NAME}/${BUCKET_DIR}/


rm -f ./${BACKUP_DIR}/${BACK_FILE}					#용량 최적화를 위해 업로드 후 원본 압축파일 삭제

#!/bin/bash

#!/bin/bash

# 사용자 설정 변수
DB_USER="std20"
DB_PASS="melt7542"           # 실제 DB 비밀번호 입력
DB_NAME="std20db"           # 백업할 데이터베이스명 입력
BACKUP_DIR="/backup"
BUCKET_NAME="std20-mysql-backup-bucket"  # 백업할 버킷 이름
BUCKET_DIR="mysql"                   # 버킷 내부의 저장 디렉토리 이름

# 날짜 포맷 정의 (예: 2026-08-06_0000)
DATE=$(date +%Y-%m-%d_%H%M)

# 백업 파일명 정의
FILE_NAME="backup-${DB_NAME}-${DATE}.sql.gz"

# 1. MySQL 백업 진행 (압축하여 용량 절약)
mysqldump -u ${DB_USER} -p${DB_PASS} ${DB_NAME} | gzip > ${BACKUP_DIR}/${FILE_NAME}

# 2. AWS S3 버킷으로 업로드
aws s3 cp ${BACKUP_DIR}/${FILE_NAME} s3://${BUCKET_NAME}/${BUCKET_DIR}/${FILE_NAME}

# 3. 서버 내부 공간 확보를 위해 로컬에 생성된 임시 백업 파일 삭제
rm -f ${BACKUP_DIR}/${FILE_NAME}
