MySQL 에서 이진타입 데이터를 저장하는것은, 파일을 저장할때 유용하긴 하지만, 효율적이지는 않다.

idx, 책이름, 출판사, 출판일, 가격.


CREATE TABLE IF NOT EXISTS tbookinfo (
	idx			int			AUTO_INCREMENT,
    bookname	VARCHAR(20)	NOT NULL,
    publisher	VARCHAR(20)	NOT NULL,
    pubdate		DATE		NOT NULL,
    price		INT			NOT NULL,
    PRIMARY KEY(idx)
);

CREATE TABLE IF NOT EXISTS tsellinfo (
	idx			INT			AUTO_INCREMENT,
    price		INT			NOT NULL		COMMENT '도서 판매가',
    salenum		INT			NOT NULL		COMMENT '도서 판매 수량',
    seldate		date		NOT NULL		COMMENT '도서 판매 일자',

이렇게 작성하면 어떤 책을 얼마에 팔았는지 알수가 없기에, 참조키를 이용한다.이러한 참조키는 이전에 만든 tbookinfo 에서 참조한다.




CREATE TABLE IF NOT EXISTS tsellinfo (
	idx			INT			AUTO_INCREMENT,
	book		INT 		NOT NULL 		COMMENT 'tbookinfo 에서 맺어진 참조 키 값',
    price		INT			NOT NULL		COMMENT '도서 판매가',
    salenum		INT			NOT NULL		COMMENT '도서 판매 수량',
    seldate		date		NOT NULL		COMMENT '도서 판매 일자',
    PRIMARY KEY(idx),
    CONSTRAINT fk_tbookinfo_idx FOREIGN KEY(book) REFERENCES tbookinfo (idx)
)COMMENT '도서 판매 정보';

mysql> desc tsellinfo;
+---------+------+------+-----+---------+----------------+
| Field   | Type | Null | Key | Default | Extra          |
+---------+------+------+-----+---------+----------------+
| idx     | int  | NO   | PRI | NULL    | auto_increment |
| book    | int  | NO   | MUL | NULL    |                |
| price   | int  | NO   |     | NULL    |                |
| salenum | int  | NO   |     | NULL    |                |
| seldate | date | NO   |     | NULL    |                |
+---------+------+------+-----+---------+----------------+


참조 키 쓰는 이유
1. 신뢰성
2. 중복 회피
